🚀 HOW TO IMPLEMENT

Step 1: Run organizer script
   php organizer.php

Step 2: Update index.php
   require 'config/autoload.php';
   use App\Controllers\HomeController;

Step 3: Test website
   http://localhost/Website

Step 4: Delete old folders (when ready)
   rm -r app/controller app/model app/views controllers models views

🔗 USE WITH NEW STRUCTURE

// OLD (scattered imports)
require 'app/controller/login_controller.php';
require 'models/BaseModel.php';
require 'helpers/SecurityHelper.php';
include 'views/home.php';

// NEW (clean & organized)
require 'config/autoload.php';
use App\Controllers\AuthController;
use App\Models\BaseModel;
$auth = new AuthController();
include VIEW_PATH . '/home/index.php';
